cmake_minimum_required(VERSION 3.8)
project(fixed_touchdown_experience)
find_package(ament_cmake REQUIRED)
find_package(ocs2_quadruped_controller REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Boost REQUIRED)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
add_executable(mechanism_experiment src/mechanism_experiment.cpp)
target_compile_features(mechanism_experiment PRIVATE cxx_std_17)
ament_target_dependencies(mechanism_experiment ocs2_quadruped_controller)
target_link_libraries(mechanism_experiment ocs2_quadruped_controller::ocs2_quadruped_controller Eigen3::Eigen Boost::boost)
add_executable(feasibility_sweep src/feasibility_sweep.cpp)
target_compile_features(feasibility_sweep PRIVATE cxx_std_17)
ament_target_dependencies(feasibility_sweep ocs2_quadruped_controller)
target_link_libraries(feasibility_sweep ocs2_quadruped_controller::ocs2_quadruped_controller Eigen3::Eigen Boost::boost)
add_executable(hull_export src/hull_export.cpp)
target_compile_features(hull_export PRIVATE cxx_std_17)
ament_target_dependencies(hull_export ocs2_quadruped_controller)
target_link_libraries(hull_export ocs2_quadruped_controller::ocs2_quadruped_controller Eigen3::Eigen Boost::boost)
install(TARGETS hull_export DESTINATION lib/${PROJECT_NAME})
install(TARGETS feasibility_sweep DESTINATION lib/${PROJECT_NAME})
install(TARGETS mechanism_experiment DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scripts DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/run_experiment.py scripts/plot_figures.py scripts/verify_data.py
  scripts/summarize_data.py scripts/prepare_sweep.py scripts/analyze_extended.py
  scripts/plot_extended.py scripts/verify_extended.py scripts/run_extended.py DESTINATION lib/${PROJECT_NAME})
ament_package()
