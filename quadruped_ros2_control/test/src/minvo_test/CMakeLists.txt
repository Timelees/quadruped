cmake_minimum_required(VERSION 3.8)
project(minvo_test)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
find_package(ament_cmake REQUIRED)
find_package(ocs2_quadruped_controller REQUIRED)
find_package(rclcpp REQUIRED)
find_package(grid_map_ros REQUIRED)
find_package(grid_map_core REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(qpoases_colcon REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
# Standalone diagnostic tests (no OCS2 dependencies)
add_executable(swing_cache_diagnostic src/swing_cache_diagnostic.cpp)
target_compile_features(swing_cache_diagnostic PRIVATE cxx_std_17)
target_link_libraries(swing_cache_diagnostic Eigen3::Eigen)
install(TARGETS swing_cache_diagnostic DESTINATION lib/${PROJECT_NAME})

foreach(target command_ramp_regression nominal_control_regression minvo_scene minvo_planning_test swing_qp_regression minvo_stability_test flat_terrain_diagnostic workspace_behavior_sim real_workspace_simulation reference_protection_test swing_obstacle_bench)
  add_executable(${target} src/${target}.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  ament_target_dependencies(${target} ocs2_quadruped_controller qpoases_colcon grid_map_core rclcpp grid_map_ros visualization_msgs)
  target_link_libraries(${target} ocs2_quadruped_controller::ocs2_quadruped_controller Eigen3::Eigen yaml-cpp)
endforeach()
enable_testing()
add_test(NAME command_ramp_regression COMMAND command_ramp_regression)
add_test(NAME minvo_planning_test COMMAND minvo_planning_test)
add_test(NAME swing_qp_regression COMMAND swing_qp_regression)
add_test(NAME swing_continuity_regression COMMAND swing_qp_regression --continuity-only)
add_test(NAME nominal_swing_regression COMMAND swing_qp_regression --nominal-only)
# Resolve the workspace's symlinked minvo_test package before locating the robot fixture.
get_filename_component(test_source_root "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
set(robot_fixture "${test_source_root}/../../../descriptions/cyberdog_description")
if(EXISTS "${robot_fixture}/config/ocs2/task.info")
  add_test(NAME command_target_regression COMMAND command_ramp_regression
    "${robot_fixture}/config/ocs2/task.info" "${robot_fixture}/config/ocs2/reference.info")
  file(READ "${robot_fixture}/config/ocs2/task.info" nominal_task)
  foreach(planning false true)
    string(REGEX REPLACE "useTerrainPlanning[ \t]+(true|false)" "useTerrainPlanning ${planning}" task "${nominal_task}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/control_${planning}.info" "${task}")
    add_test(NAME control_problem_${planning} COMMAND nominal_control_regression
      "${CMAKE_CURRENT_BINARY_DIR}/control_${planning}.info"
      "${robot_fixture}/urdf/robot.urdf" "${robot_fixture}/config/ocs2/reference.info")
  endforeach()
endif()

add_test(NAME minvo_stability_test COMMAND minvo_stability_test)
foreach(scene single pillars stairs)
  add_test(NAME scene_${scene} COMMAND minvo_scene ${CMAKE_CURRENT_SOURCE_DIR}/config/${scene}.yaml ${CMAKE_CURRENT_BINARY_DIR}/data/${scene} --export-only)
endforeach()
install(TARGETS nominal_control_regression minvo_scene minvo_planning_test swing_qp_regression minvo_stability_test flat_terrain_diagnostic workspace_behavior_sim real_workspace_simulation reference_protection_test swing_obstacle_bench DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch script DESTINATION share/${PROJECT_NAME})
ament_package()
