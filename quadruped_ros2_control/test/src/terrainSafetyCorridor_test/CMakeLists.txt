cmake_minimum_required(VERSION 3.8)
project(terrain_safety_corridor_test)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
find_package(ament_cmake REQUIRED)
find_package(ocs2_quadruped_controller REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(grid_map_ros REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(terrain_corridor_test src/terrain_corridor_test.cpp)
target_compile_features(terrain_corridor_test PRIVATE cxx_std_17)
target_compile_options(terrain_corridor_test PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(terrain_corridor_test ocs2_quadruped_controller rclcpp grid_map_ros visualization_msgs ament_index_cpp)
target_link_libraries(terrain_corridor_test ocs2_quadruped_controller::ocs2_quadruped_controller Eigen3::Eigen)
enable_testing()
add_test(NAME terrain_corridor_geometry_and_pipeline COMMAND terrain_corridor_test --verify-only)
install(TARGETS terrain_corridor_test DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS plot_corridor_sections.py plot_corridors.py plot_evidence.py analyze_stability.py runtime_probe.py scenario_probe.py run_trial.py DESTINATION lib/${PROJECT_NAME})
ament_package()
